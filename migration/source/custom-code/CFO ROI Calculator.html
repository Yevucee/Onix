function applyRate(rate, source, date) {

  liveUsdRate = rate;

  rateSource = source;

  // Only update the placeholder if the user hasn't typed their own value

  const input = document.getElementById('advUsdRate');

(function fetchLiveRate() {

  // 1. Try cache first

  try {

    const cached = localStorage.getItem(CACHE_KEY);

    if (cached) {

      const p = JSON.parse(cached);

      if (Date.now() - p.timestamp < CACHE_TTL && p.rate) {

        applyRate(p.rate, 'cached', new Date(p.timestamp));

        // Still try to refresh in background if cache is older than 1 hour

        if (Date.now() - p.timestamp > 60 * 60 * 1000) {

          fetchFromAPI(true); // silent background refresh

        }

        return;

      }

    }

  } catch (e) {}


  // 2. No valid cache — fetch live

  fetchFromAPI(false);

})();


function fetchFromAPI(silent) {

  // Three APIs tried in sequence — first success wins.

  // API 1: fawazahmed0/exchange-api via jsDelivr (no key, no rate limit)

  // API 2: fawazahmed0/exchange-api via Cloudflare fallback

  // API 3: open.er-api.com open endpoint

  const apis = [

    {

      url: 'https://cdn.jsdelivr.net/npm/@fawazahmed0/currency-api@latest/v1/currencies/usd.json',

      extract: data => {

        const rate = data?.usd?.ghs;

        if (!rate) throw new Error('No GHS in response');

        return Math.round(rate * 100) / 100;

      }

    },

    {

      url: 'https://latest.currency-api.pages.dev/v1/currencies/usd.json',

      extract: data => {

        const rate = data?.usd?.ghs;

        if (!rate) throw new Error('No GHS in response');

        return Math.round(rate * 100) / 100;

      }

    },

    {

      url: 'https://open.er-api.com/v6/latest/USD',

      extract: data => {

        if (data?.result !== 'success' || !data.rates?.GHS) throw new Error('Invalid response');

        return Math.round(data.rates.GHS * 100) / 100;

      }

    }

  ];


  function tryNext(index) {

    if (index >= apis.length) {

      if (!silent) applyRate(P.usdToGhs, 'fallback', null);

      return;

    }

    const api = apis[index];

    fetch(api.url)

      .then(r => { if (!r.ok) throw new Error('HTTP ' + r.status); return r.json(); })

      .then(data => {

        const rate = api.extract(data);

        try {

          localStorage.setItem(CACHE_KEY, JSON.stringify({ rate, timestamp: Date.now() }));

        } catch (e) {}

        applyRate(rate, 'live', new Date());

      })

      .catch(() => tryNext(index + 1));

  }


  tryNext(0);

}

  const spaceCostM2Val = numVal('spaceCostM2', P.defaultSpaceCostM2);

  const annualSpaceCost = spaceRequired * spaceCostM2Val * 12;


  const annualStaffCost = staffHoursYear * labourRate;


  const capexTotal = r * capexPerRack;

  const annualCapex = capexTotal / P.defaultAmortisationYears;

  const maintenanceCost = capexTotal * maintRate;


  const facilityCapexVal = numVal('facilityCapex', P.defaultFacilityCapex);

  const annualFacilityCapex = facilityCapexVal / P.facilityAmortisationYears;


  const genHours = numVal('genRunHours', P.defaultGenRunHours);

  const genCostPerHr = numVal('genFuelCost', P.defaultGenFuelCostPerHour);

  const annualGenFuel = genHours * genCostPerHr;


  const inHouseTotal = powerCost + annualSpaceCost + annualStaffCost + maintenanceCost + annualCapex + annualFacilityCapex + annualGenFuel;


  // ─── Downtime ───

  const downtimeSens = document.getElementById('downtimeSens').value;

  const downtimeCostPerHour = P.downtimeSensitivity[downtimeSens];

  const dtHours = parseFloat(document.getElementById('downtimeHours').value) || 16;

  const downtimeLoss = dtHours * downtimeCostPerHour;

  const riskAdjustedInHouse = inHouseTotal + downtimeLoss;


  // ─── Onix costs (USD → GH₵) ───

  const rackPriceUSD = kw >= 5 ? P.rack5kw : P.rack3kw;

  const packageCap = kw >= 5 ? r * 5 : r * 3;

  const extraKw = Math.max(0, itLoad - packageCap);

  const extraPowerUSD = extraKw * P.extraKwPerMonth;


  const rackCostUSD = r * rackPriceUSD * 12;

  const ccMrcUSD = ccQty * P.crossConnectMRC * 12;

  const ccNrcUSD = ccQty * P.crossConnectNRC;

  const extraPowerAnnualUSD = extraPowerUSD * 12;


  const onixTotalUSD = rackCostUSD + ccMrcUSD + extraPowerAnnualUSD;

  const onixTotal = onixTotalUSD * usdRate;


  const cageNRE = segregation ? r * P.cageNREPerRack : 0;

  const oneOffUSD = ccNrcUSD + cageNRE;

  const oneOff = oneOffUSD * usdRate;


  const savings = inHouseTotal - onixTotal;

  const riskSavings = riskAdjustedInHouse - onixTotal;

  const payback = savings > 0 && oneOff > 0 ? oneOff / (savings / 12) : null;


  // ─── Render Summary Cards ───

  document.getElementById('summaryCards').innerHTML = `

    <div class="result-card">

      <p class="rlabel">In-House Annual Cost</p>

      <p class="rvalue">${ghsFmt(inHouseTotal)}</p>

    </div>

    <div class="result-card">

      <p class="rlabel">Onix Annual Cost</p>

      <p class="rvalue">${ghsFmt(onixTotal)}</p>

    </div>

    <div class="result-card ${savings >= 0 ? 'rc-savings' : 'rc-cost'}">

      <p class="rlabel">${savings >= 0 ? 'Estimated Annual Savings' : 'Estimated Annual Difference'}</p>

      <p class="rvalue">${ghsFmt(Math.abs(savings))}</p>

      <p class="rsub">${savings >= 0 ? 'You save with Onix' : 'In-house is cheaper'}</p>

    </div>

    <div class="result-card rc-highlight">

      <p class="rlabel">Payback Period</p>

      <p class="rvalue">${payback !== null ? Math.ceil(payback) + ' months' : 'N/A'}</p>

      <p class="rsub">${payback !== null ? 'On ' + ghsFmt(oneOff) + ' one-off costs' : (savings <= 0 ? 'No savings to offset' : 'No one-off costs')}</p>

    </div>

  `;


  // ─── Downtime Cards ───

  document.getElementById('downtimeCards').innerHTML = `

    <div class="result-card rc-cost">

      <p class="rlabel">Downtime Losses (In-House)</p>

      <p class="rvalue">${ghsFmt(downtimeLoss)}</p>

      <p class="rsub">${dtHours} hrs × ${ghsFmt(downtimeCostPerHour)}/hr</p>

    </div>

    <div class="result-card ${riskSavings >= 0 ? 'rc-savings' : 'rc-cost'}">

      <p class="rlabel">Risk-Adjusted Savings</p>

      <p class="rvalue">${ghsFmt(Math.abs(riskSavings))}</p>

      <p class="rsub">${riskSavings >= 0 ? 'Including downtime impact' : 'Onix is more expensive even with downtime'}</p>

    </div>

  `;


  // ─── Breakdown Chart ───

  const ihBreakdown = [

    { name: 'Power', value: powerCost },

    { name: 'Space', value: annualSpaceCost },

    { name: 'Staff', value: annualStaffCost },

    { name: 'Maintenance', value: maintenanceCost },

    { name: 'Server CapEx', value: annualCapex },

    { name: 'Facility CapEx', value: annualFacilityCapex },

    { name: 'Generator Fuel', value: annualGenFuel },

    { name: 'Downtime', value: downtimeLoss },

  ];

  const onixBreakdown = [

    { name: 'Rack Cost', value: rackCostUSD * usdRate },

    { name: 'Cross-Connect', value: ccMrcUSD * usdRate },

    { name: 'Addtl Power', value: extraPowerAnnualUSD * usdRate },

  ];

  const allNames = [...new Set([...ihBreakdown.map(i=>i.name), ...onixBreakdown.map(i=>i.name)])];

  const maxVal = Math.max(

    ihBreakdown.reduce((s,i) => s+i.value, 0),

    onixBreakdown.reduce((s,i) => s+i.value, 0)

  );


  function renderBar(label, items, total) {

    const segs = items.filter(i => i.value > 0).map(i => {

      const ci = allNames.indexOf(i.name);

      const pct = maxVal > 0 ? (i.value / maxVal * 100) : 0;

      return `<div class="bar-segment" style="width:${pct}%;background:${COLORS[ci % COLORS.length]}"><span class="bar-tip">${i.name}: ${ghsFmt(i.value)}</span></div>`;

    }).join('');

    return `<div class="bar-row">

      <span class="bar-label">${label}</span>

      <div class="bar-track">${segs}</div>

      <span class="bar-total">${ghsFmt(total)}</span>

    </div>`;

  }


  const legendHtml = allNames.map((n, i) => `<div class="legend-item"><div class="legend-swatch" style="background:${COLORS[i % COLORS.length]}"></div>${n}</div>`).join('');


  document.getElementById('chartBody').innerHTML = `

    <div class="breakdown-chart">

      ${renderBar('In-House', ihBreakdown, ihBreakdown.reduce((s,i)=>s+i.value,0))}

      ${renderBar('Onix', onixBreakdown, onixBreakdown.reduce((s,i)=>s+i.value,0))}

    </div>

    <div class="legend">${legendHtml}</div>

  `;


  // ─── Config Summary ───

  const rackPkg = kw >= 5 ? '5 kW' : '3 kW';

  document.getElementById('configGrid').innerHTML = `

    <div class="config-item"><p class="ci-label">Racks</p><p class="ci-value">${r}</p></div>

    <div class="config-item"><p class="ci-label">Rack Package</p><p class="ci-value">${rackPkg}</p></div>

    <div class="config-item"><p class="ci-label">Total IT Load</p><p class="ci-value">${itLoad} kW</p></div>

    <div class="config-item"><p class="ci-label">Cross-Connects</p><p class="ci-value">${ccQty}</p></div>

  `;


  // ─── Assumptions ───

  const rateLabel = advUsdVal ? `${usdRate} (manual)` : `${usdRate} (${rateSource})`;

  document.getElementById('assumptionsGrid').innerHTML = `

    <div><span class="al">Rack density:</span> <span class="av">${kw} kW</span></div>

    <div><span class="al">PUE:</span> <span class="av">${pue}</span></div>

    <div><span class="al">Space factor:</span> <span class="av">${m2Factor} m²/rack</span></div>

    <div><span class="al">Maintenance:</span> <span class="av">${(maintRate*100).toFixed(0)}%</span></div>

    <div><span class="al">Staff hrs/yr:</span> <span class="av">${staffHoursYear.toLocaleString()}</span></div>

    <div><span class="al">Labour rate:</span> <span class="av">GH₵${labourRate}/hr</span></div>

    <div><span class="al">Tariff:</span> <span class="av">GH₵${tariffVal}/kWh</span></div>

    <div><span class="al">USD rate:</span> <span class="av">1 USD = ${rateLabel} GH₵</span></div>

    <div><span class="al">Facility CapEx:</span> <span class="av">${ghsFmt(facilityCapexVal)}</span></div>

    <div><span class="al">Gen hours/yr:</span> <span class="av">${genHours.toLocaleString()}</span></div>

    <div><span class="al">Gen fuel:</span> <span class="av">GH₵${genCostPerHr}/hr</span></div>

    <div><span class="al">HW amortisation:</span> <span class="av">${P.defaultAmortisationYears} years</span></div>

  `;

  document.getElementById('adjustLink').style.display = advOpen ? 'none' : 'block';


  // ─── Disclaimer ───

  const sourceLabel = rateSource === 'live' ? 'live' : rateSource === 'cached' ? 'cached' : 'default';

  document.getElementById('disclaimerRate').textContent = `Onix pricing is converted at ${usdRate} GH₵ per USD (${sourceLabel} rate).`;


  el.classList.add('visible');

}
